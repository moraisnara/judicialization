\section{Motivation}
% ============================================================

\begin{frame}{Does Judicial Intervention Shape Electoral Competition?}
\begin{columns}[T]
\begin{column}{0.58\textwidth}
\textbf{Research question:}\\
Does pre-election judicial intervention causally affect the competitive dynamics of mayoral races?
\bigskip

\textbf{The mechanism:}
\begin{itemize}
  \item Candidates are sued in electoral courts \emph{before} election day
  \item Lawsuits impose legal costs, reputational exposure, and campaign disruption
  \item This could shift vote shares, alter entry decisions, and change voter behavior
\end{itemize}
\bigskip

\textbf{The challenge:}\\
Close races \emph{attract} more lawsuits — endogeneity goes in both directions
\end{column}
\begin{column}{0.40\textwidth}
\begin{block}{This Paper}
  \begin{itemize}
    \item 5,570 Brazilian municipalities
    \item 2020 and 2024 mayoral elections
    \item TSE administrative data: judicial records $+$ electoral results
    \item Identification: \textbf{Bartik shift-share IV}
  \end{itemize}
\end{block}
\begin{alertblock}{Key Findings}
  Strong first stage (F~$\approx$~46). Competition effects: directionally consistent, imprecisely estimated. Voter behavior and incumbency: precise nulls.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Why Brazil? Why Municipalities?}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Brazil's Electoral Justice System:}
\begin{itemize}
  \item \emph{Tribunal Superior Eleitoral} (TSE) + 27 \emph{Tribunais Regionais Eleitorais} (TREs) + 2,800+ first-instance zones
  \item Adjudicates eligibility, campaign finance, conduct \emph{before} election day
  \item Tens of thousands of cases per cycle — among the world's most active electoral courts
  \item Rich variation: topic mix differs by municipality, state, and year
\end{itemize}
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Municipal Setting}
  \begin{itemize}
    \item 5,570 simultaneous contests $=$ large sample
    \item Homogeneous office: directly elected mayor
    \item Uniform electoral rules (Constitution Art.~29)
    \item Comparable competitive dynamics across municipalities
  \end{itemize}
\end{block}
\begin{block}{Two Election Cycles}
  2020 (Nov 15) $\to$ \textbf{shares}\\
  2024 (Oct 6) $\to$ \textbf{treatment + outcomes}
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Institutional Background}
% ============================================================

\begin{frame}{Brazil's Electoral Justice System}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Three-tier judicial structure:}
\begin{enumerate}
  \item \textbf{Zona Eleitoral (ZE)} — first instance; 2,800+ nationwide
  \item \textbf{Tribunal Regional Eleitoral (TRE)} — 27 state appellate courts
  \item \textbf{Tribunal Superior Eleitoral (TSE)} — national apex
\end{enumerate}
\bigskip
\textbf{Pre-election jurisdiction:}
\begin{itemize}
  \item Candidate registration challenges (\emph{impugnaç\~{a}o})
  \item Abuse of economic/political power
  \item Campaign finance violations
  \item Prohibited propaganda; vote-buying
  \item Candidate conduct (\emph{conduta vedada})
\end{itemize}
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Election Timeline}
  \footnotesize
  \begin{enumerate}
    \item Candidate registration (August)
    \item \textbf{Pre-election lawsuits} (Aug--Oct)
    \item \textbf{Election Day} (Oct/Nov)
    \item Post-election proceedings
    \item Cassation if ruling goes against winner
  \end{enumerate}
\end{block}
\bigskip
\footnotesize
\textbf{2020:} Election day = November 15\\
\textbf{2024:} Election day = October 6\\[4pt]
We use only cases filed \emph{before} these dates.
\end{column}
\end{columns}
\end{frame}


\begin{frame}{What Lawsuits Do — and Who Files Them}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Effects on targeted candidates:}
\begin{itemize}
  \item \textbf{Candidacy at risk:} court may rule ineligible before election day
  \item \textbf{Reputational cost:} public filing, media coverage, stigma
  \item \textbf{Resource diversion:} legal defense competes with campaigning
  \item \textbf{Quality signal to voters:} a respondent may be perceived as ethically challenged — or as a victim of political harassment
\end{itemize}
\bigskip
\textbf{Who files?}\\
Opposing candidates, parties, \emph{Ministério Público Eleitoral}, any citizen
\end{column}
\begin{column}{0.43\textwidth}
\begin{alertblock}{Treatment Variable}
  \textbf{Passive-party (polo passivo) competition-relevant cases:}
  \begin{itemize}
    \item Mayoral candidate is the \emph{respondent}
    \item Filed before election day
    \item Subject $\neq$ routine registration (RRC)
  \end{itemize}
  \bigskip
  \textbf{Measurement:}\\
  $\Delta\log(1+\text{lawsuits}_{m})$\\from 2020 to 2024
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Competition-Relevant vs.\ Bureaucratic Cases}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
Not all electoral cases affect competition:
\bigskip
\begin{itemize}
  \item \textbf{Routine Registration Cases (RRC):}\\
  Subject code 11618 — \emph{Registro de Candidatura}\\
  Every candidacy generates one; mandatory; no adversarial content
  \item \textbf{Competition-relevant cases:}\\
  Substantive challenges that could alter the race; filed by opponents, MPE, or citizens
\end{itemize}
\bigskip
\alert{RRC excluded from both the treatment and the Bartik instrument}
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Classification Table}
  \footnotesize
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{lc}
    \toprule
    Case subject & Included? \\
    \midrule
    Abuso de Poder & \tick \\
    Captaç\~{a}o Il\'{i}cita de Sufr\'{a}gio & \tick \\
    Propaganda Irregular & \tick \\
    Conduta Vedada & \tick \\
    Uso Indevido de Meios & \tick \\
    Financiamento de Campanha & \tick \\
    \midrule
    Registro de Candidatura (11618) & \cross \\
    \bottomrule
  \end{tabular}
\end{block}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Pre-Election Timing Restriction}
\begin{itemize}
  \item \textbf{Restriction:} we keep only cases filed \textit{strictly before} election day:
  \[
    \texttt{DT\_AUTU\!A\c{C}\!\~{A}O} < \text{election date}
  \]
  \medskip
  \item \textbf{Rationale:} only pre-election cases can plausibly affect the upcoming competitive dynamics through legal costs, reputational signaling, or candidacy threats
  \medskip
  \item Post-election \emph{cassação} proceedings determine whether the \emph{realized winner} is removed — a distinct phenomenon that could reverse causality
  \medskip
  \item This restriction is applied identically to both the 2020 baseline (for shares) and the 2024 measurement (for treatment and shifts)
\end{itemize}
\end{frame}


\begin{frame}{The Endogeneity Problem}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{OLS is likely biased:}
\bigskip
\begin{enumerate}
  \item \textbf{Competition $\to$ lawsuits}\\
  Close races attract more suits (parties fight harder); positive confound with competitiveness
  \medskip
  \item \textbf{Vulnerability $\to$ lawsuits}\\
  Incumbent weakness invites challengers to file; negative confound with incumbency survival
  \medskip
  \item \textbf{Unobservables}\\
  Local political culture, partisan intensity, lawyer density — correlated with both
\end{enumerate}
\bigskip
Direction of bias: \textbf{ambiguous}
\end{column}
\begin{column}{0.43\textwidth}
\begin{alertblock}{Identification Need}
  We need exogenous variation in the \emph{level} of judicialization that is independent of the municipality's own competitive dynamics in 2024.
\end{alertblock}
\bigskip
\begin{block}{Our Strategy}
  \textbf{Bartik shift-share IV:}\\
  Predicted 2024 lawsuits based on the municipality's 2020 topic mix interacted with how each topic grew statewide — a source of variation the municipality itself does not drive.
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Identification Strategy}
% ============================================================

\begin{frame}{Bartik Shift-Share — Intuition}
\begin{columns}[T]
\begin{column}{0.57\textwidth}
\textbf{In 2020, each municipality has a topic mix:}\\
Some had mostly ``abuse of power'' cases; others had more ``campaign finance'' or ``prohibited propaganda'' cases.
\bigskip

\textbf{By 2024, topics grew at different rates statewide.}\\
If ``abuse of power'' lawsuits surged in Minas Gerais between 2020 and 2024, then a municipality that had many ``abuse of power'' cases in 2020 is \emph{more exposed} to this wave — regardless of its own 2024 politics.
\bigskip

\textbf{The instrument:}\\
$\bZ$ = municipality's predicted 2024 lawsuits from its 2020 topic mix $\times$ state-level growth in each topic
\end{column}
\begin{column}{0.41\textwidth}
\begin{alertblock}{Key Design Logic}
  \begin{enumerate}
    \item \textbf{Shares}: from 2020 — pre-determined, before 2024 politics materialize
    \item \textbf{Shifts}: state-level — driven by systemic factors the municipality cannot individually control
    \item \textbf{Product}: predicted 2024 exposure that is exogenous to the municipality's own 2024 competitive dynamics
  \end{enumerate}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Bartik Instrument — Formal Construction}
\[
\bZ = \sum_{k \notin \text{RRC}} s_{m,k,2020} \;\cdot\; g_{\text{UF}(m),\,k,\,-m}
\]
\bigskip
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\textwidth}{L{2.4cm}X}
  $s_{m,k,2020}$ & Municipality $m$'s share of topic $k$ among all non-RRC lawsuits in 2020; re-normalized after RRC exclusion so that $\sum_k s_{m,k,2020} = 1$ \\
  $g_{\text{UF},k,-m}$ & Leave-one-out (LOO) log growth of topic $k$ in state UF$(m)$ from 2020 to 2024:
  $\displaystyle g = \log\!\!\left(\frac{T_{\text{UF},k,2024} - C_{m,k,2024}}{T_{\text{UF},k,2020} - C_{m,k,2020}}\right)$ \\
  $k$ & Topic cell = DS\_CLASSE $\times$ DS\_ASSUNTO\_PRINCIPAL; $\sim$100+ distinct cells \\
  \text{LOO} & Municipality $m$ excluded from its own state shift to prevent mechanical correlation \\
  \text{NaN} & $g = \text{NaN}$ when the LOO denominator $\leq 0$; those topic-municipality pairs excluded from the sum \\
\end{tabularx}
\end{frame}


\begin{frame}{Shares: 2020 Topic Composition}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Construction:}
\begin{enumerate}
  \item Count unique non-RRC cases per (municipality, topic) in 2020
  \item $s_{m,k,2020} = C_{m,k,2020} \,/\, \sum_{k} C_{m,k,2020}$
  \item Shares sum to 1.0 per municipality by construction
  \item Municipalities with zero non-RRC 2020 cases: excluded (no valid shares)
\end{enumerate}
\bigskip
\textbf{Granularity:} DS\_CLASSE $\times$ DS\_ASSUNTO\_PRINCIPAL gives $\sim$100+ topic cells, capturing fine-grained legal taxonomy
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Why 2020 Shares?}
  Shares are \textbf{pre-determined}: based on the 2020 case distribution, two electoral cycles before the 2024 election dynamics unfold.
  \medskip\\
  Exposure $s_{m,k,2020}$ cannot be chosen or manipulated by actors anticipating 2024 competition — a core validity argument.
\end{block}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Shifts: State-Level Log Growth, Leave-One-Out}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Construction:}
\begin{enumerate}
  \item Compute state $\times$ topic totals: $T_{\text{UF},k,t}$ for $t \in \{2020, 2024\}$
  \item Remove $m$'s own contribution (LOO): subtract $C_{m,k,t}$
  \item Compute log growth; set NaN if LOO denominator $\leq 0$
\end{enumerate}
\bigskip
\textbf{Shift level: state (UF)}
\begin{itemize}
  \item Closer geographic comparison than national shifts
  \item Absorbs national trends that are common to all municipalities in the same state
  \item 27 states gives meaningful sample for state-level trends
\end{itemize}
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Design Choices Defended}
  \begin{description}
    \item[\textbf{LOO}] Prevents $m$ from mechanically driving its own shift in states with few municipalities
    \item[\textbf{RRC out}] Shifts computed on substantive litigation only; routine registration noise excluded
    \item[\textbf{Log}] Proportional changes; symmetric treatment of growth and decline
  \end{description}
\end{block}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Exclusion Restriction}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Formal condition:}\\
$E[\bZ \cdot u_m \mid \mathbf{X}_m, \delta_{\text{UF}}] = 0$
\bigskip

\textbf{Core argument:}\\
State-wide growth in topic $k$ lawsuits is driven by:
\begin{itemize}
  \item TRE jurisprudence shifts
  \item National party strategies
  \item Legislative changes
  \item MPE enforcement waves
\end{itemize}
These systemic factors are orthogonal to the specific 2024 competitive environment in any single small municipality, conditional on state FE and baseline controls.
\end{column}
\begin{column}{0.43\textwidth}
\begin{alertblock}{Potential Threats}
  \begin{enumerate}
    \item \textbf{Large-municipality dominance}: municipality $m$ drives state shift\\
    $\rightarrow$ mitigated by LOO; most municipalities are small
    \medskip
    \item \textbf{2020 shares reflect 2020 competition}: shares correlated with unobserved persistence\\
    $\rightarrow$ conditional on state FE + 2020 competitive controls
    \medskip
    \item \textbf{Topic-specific shocks}: state-level growth in a topic correlated with state-level political shock that also affects $m$\\
    $\rightarrow$ state FE absorbs
  \end{enumerate}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Instrument Validity Checks}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{We verify:}
\begin{enumerate}
  \item \textbf{Relevance:} first-stage F-statistic (reported below); target $\geq 10$; actual $\approx 46$
  \medskip
  \item \textbf{Balance:} regress $\bZ$ on each pre-determined covariate (one at a time, state FE); expect $p > 0.10$ for all
  \medskip
  \item \textbf{Distribution:} no implausible outliers in $\bZ$ after residualization on state FE
  \medskip
  \item \textbf{Stability:} first-stage coefficient stable across control sets, samples, and FE granularity
\end{enumerate}
\end{column}

\[

\[
\Delta Y_m \;=\; \alpha + \tau\, \widehat{\Delta\log(1+\ell_{m})} + \mathbf{X}_m'\boldsymbol{\gamma} + \delta_{\text{UF}(m)} + u_m

\end{itemize}

\section{Conclusion}
% ============================================================

\begin{frame}{Summary of Findings}
\begin{columns}[T]
\begin{column}{0.57\textwidth}
\begin{enumerate}
  \item \textbf{Strong identification:}\\
  Bartik IV is highly relevant (F~$\approx$~50, stable across samples). First-stage coefficient $\approx$1.51 regardless of control set or FE level.
  \bigskip
  \item \textbf{Electoral competition:}\\
  Directionally consistent winner consolidation ($\Delta$margin $+$5 pp; $\Delta$runner-up $-$2.2 pp; $\Delta$winner $+$2.8 pp) — but effects are imprecisely estimated ($p \approx 0.15$--$0.23$).
  \bigskip
  \item \textbf{Voter behavior:}\\
  Turnout and null votes: precise nulls. Blank votes: positive, $p = 0.12$.
  \bigskip
  \item \textbf{Incumbency and extended outcomes:}\\
  All null across incumbent ran, won, party switch, ideology, volatility, gender entry.
\end{enumerate}
\end{column}
\begin{column}{0.41\textwidth}
\begin{alertblock}{Main Takeaway}
  We provide credibly identified estimates of the electoral effects of judicialization using a Bartik shift-share instrument.
  \medskip\\
  The estimates are directionally interpretable — suggesting winner consolidation — but Brazilian electoral litigation, as measured here, does not produce statistically detectable changes in who wins, voter turnout, or incumbency survival.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Contributions and Next Steps}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Contributions:}
\begin{itemize}
  \item First Bartik IV applied to electoral judicialization; establishes a blueprint for identifying causal electoral effects of litigation
  \item Large-scale evidence from 5,570 Brazilian municipalities across two election cycles
  \item Separates bureaucratic (RRC) from competition-relevant litigation — a methodological contribution to measurement
  \item Rich outcome battery covering competition, voter behavior, incumbency, ideology, and gender
\end{itemize}
\end{column}

    \item Case \emph{outcomes} (ruling in favor vs.\ against) as a second instrument
    \item Second-round analysis for 222 large-city municipalities
    \item Candidate entry structural model
    \item Longer panel: 2012--2024 for trend identification
  \end{itemize}
\end{block}
