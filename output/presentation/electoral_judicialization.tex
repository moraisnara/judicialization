\section{Motivation}
% ============================================================

\begin{frame}{Does Judicial Intervention Shape Electoral Competition?}
\begin{columns}[T]
\begin{column}{0.58\textwidth}
\textbf{Research question:}\\
Does pre-election judicial intervention causally affect the competitive dynamics of mayoral races?
\bigskip

\textbf{The mechanism:}
\begin{itemize}
  \item Candidates are sued in electoral courts \emph{before} election day
  \item Lawsuits impose legal costs, reputational exposure, and campaign disruption
  \item This could shift vote shares, alter entry decisions, and change voter behavior
\end{itemize}
\bigskip

\textbf{The challenge:}\\
Close races \emph{attract} more lawsuits — endogeneity goes in both directions
\end{column}
\begin{column}{0.40\textwidth}
\begin{block}{This Paper}
  \begin{itemize}
    \item 5,560 Brazilian municipalities
    \item 2020 and 2024 mayoral elections
    \item TSE administrative data: judicial records $+$ electoral results
    \item Identification: \textbf{Bartik shift-share IV}
  \end{itemize}
\end{block}
\begin{alertblock}{Key Findings}
  Two instrument variants (F\,$\approx$\,10--20). Directionally consistent winner consolidation; blank vote rate rises significantly under the broader specification. Voter participation: precise nulls.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Why Brazil? Why Municipalities?}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Brazil's Electoral Justice System:}
\begin{itemize}
  \item \emph{Tribunal Superior Eleitoral} (TSE) + 27 \emph{Tribunais Regionais Eleitorais} (TREs) + 2,800+ first-instance zones
  \item Adjudicates eligibility, campaign finance, conduct \emph{before} election day
  \item Tens of thousands of cases per cycle — among the world's most active electoral courts
  \item Rich variation: topic mix differs by municipality, state, and year
\end{itemize}
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Municipal Setting}
  \begin{itemize}
    \item 5,570 simultaneous contests $=$ large sample
    \item Homogeneous office: directly elected mayor
    \item Uniform electoral rules (Constitution Art.~29)
    \item Comparable competitive dynamics across municipalities
  \end{itemize}
\end{block}
\begin{block}{Two Election Cycles}
  2020 (Nov 15) $\to$ \textbf{shares}\\
  2024 (Oct 6) $\to$ \textbf{treatment + outcomes}
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Institutional Background}
% ============================================================

\begin{frame}{Brazil's Electoral Justice System}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Three-tier judicial structure:}
\begin{enumerate}
  \item \textbf{Zona Eleitoral (ZE)} — first instance; 2,800+ nationwide
  \item \textbf{Tribunal Regional Eleitoral (TRE)} — 27 state appellate courts
  \item \textbf{Tribunal Superior Eleitoral (TSE)} — national apex
\end{enumerate}
\bigskip
\textbf{Pre-election jurisdiction:}
\begin{itemize}
  \item Candidate registration challenges (\emph{impugna\c{c}\~{a}o})
  \item Abuse of economic/political power
  \item Campaign finance violations
  \item Prohibited propaganda; vote-buying
  \item Candidate conduct (\emph{conduta vedada})
\end{itemize}
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Election Timeline}
  \footnotesize
  \begin{enumerate}
    \item Candidate registration (August)
    \item \textbf{Pre-election lawsuits} (Aug--Oct)
    \item \textbf{Election Day} (Oct/Nov)
    \item Post-election proceedings
    \item Cassation if ruling goes against winner
  \end{enumerate}
\end{block}
\bigskip
\footnotesize
\textbf{2020:} Election day = November 15\\
\textbf{2024:} Election day = October 6\\[4pt]
We use only cases filed \emph{before} these dates.
\end{column}
\end{columns}
\end{frame}


\begin{frame}{What Lawsuits Do --- and Who Files Them}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Effects on targeted candidates:}
\begin{itemize}
  \item \textbf{Candidacy at risk:} court may rule ineligible before election day
  \item \textbf{Reputational cost:} public filing, media coverage, stigma
  \item \textbf{Resource diversion:} legal defense competes with campaigning
  \item \textbf{Quality signal to voters:} a respondent may be perceived as ethically challenged --- or as a victim of political harassment
\end{itemize}
\bigskip
\textbf{Who files?}\\
Opposing candidates, parties, \emph{Ministério Público Eleitoral}, any citizen
\end{column}
\begin{column}{0.43\textwidth}
\begin{alertblock}{Treatment Variable}
  \textbf{Adversarial competition-relevant cases:}
  \begin{itemize}
    \item Mayoral candidate is the \emph{respondent}
    \item Filed before election day
    \item Substantive challenge (not mandatory administrative filing)
  \end{itemize}
  \bigskip
  \textbf{Measurement:}\\
  $\Delta\log(1+\text{lawsuits}_{m})$\\from 2020 to 2024
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Administrative vs.\ Adversarial Electoral Litigation}
\begin{columns}[T]
\begin{column}{0.57\textwidth}
Not all electoral case subjects involve genuine legal challenges to candidates:
\bigskip
\begin{description}
  \item[\textbf{Candidacy Registration Confirmation (RRC):}] Automatically generated for \emph{every} registered candidacy. No plaintiff, no adversarial content, no legal risk. Purely administrative processing.
  \medskip
  \item[\textbf{Coalition/Party Slate Registration (DRAP):}] Filed by each party or coalition before every election; mandatory under electoral law. No opposing party, no discretion. Universal coverage ($>$97\% of municipalities in 2020).
  \medskip
  \item[\textbf{Adversarial challenges:}] Filed by opponents, the Electoral Public Prosecutor (\emph{MPE}), or citizens against a specific candidate's conduct, eligibility, or campaign behaviour. These are the cases that constitute genuine judicial intervention.
\end{description}
\end{column}
\begin{column}{0.41\textwidth}
\begin{block}{Case Volume (2020)}
  \footnotesize
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{lrc}
    \toprule
    Type & Volume & Incl.? \\
    \midrule
    Candidacy registration (RRC) & $\sim$994k & \cross \\
    Coalition registration (DRAP) & $\sim$137k & \cross \\
    \midrule
    Eligibility challenges & & \tick \\
    Abuse of power & & \tick \\
    Campaign conduct & & \tick \\
    Propaganda violations & & \tick \\
    Information violations & & \tick \\
    \bottomrule
  \end{tabular}
\end{block}
\medskip
\begin{alertblock}{Instrument scope}
  \footnotesize
  We test two variants: one that excludes only the mandatory filings, and one (preferred) that retains only adversarial subjects.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Pre-Election Timing Restriction}
\begin{itemize}
  \item \textbf{Restriction:} we keep only cases filed \textit{strictly before} election day:
  \[
    \text{filing date} < \text{election date}
  \]
  \medskip
  \item \textbf{Rationale:} only pre-election cases can plausibly affect the upcoming competitive dynamics through legal costs, reputational signaling, or candidacy threats
  \medskip
  \item Post-election \emph{cassa\c{c}\~{a}o} proceedings determine whether the \emph{realized winner} is removed — a distinct phenomenon that could reverse causality
  \medskip
  \item This restriction is applied identically to both the 2020 baseline (for shares) and the 2024 measurement (for treatment and shifts)
\end{itemize}
\end{frame}


\begin{frame}{The Endogeneity Problem}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{OLS is likely biased:}
\bigskip
\begin{enumerate}
  \item \textbf{Competition $\to$ lawsuits}\\
  Close races attract more suits (parties fight harder); positive confound with competitiveness
  \medskip
  \item \textbf{Vulnerability $\to$ lawsuits}\\
  Incumbent weakness invites challengers to file; negative confound with incumbency survival
  \medskip
  \item \textbf{Unobservables}\\
  Local political culture, partisan intensity, lawyer density — correlated with both
\end{enumerate}
\bigskip
Direction of bias: \textbf{ambiguous}
\end{column}
\begin{column}{0.43\textwidth}
\begin{alertblock}{Identification Need}
  We need exogenous variation in the \emph{level} of judicialization that is independent of the municipality's own competitive dynamics in 2024.
\end{alertblock}
\bigskip
\begin{block}{Our Strategy}
  \textbf{Bartik shift-share IV:}\\
  Predicted 2024 lawsuits based on the municipality's 2020 topic mix interacted with how each topic grew statewide — a source of variation the municipality itself does not drive.
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Identification Strategy}
% ============================================================

\begin{frame}{Bartik Shift-Share --- Intuition}
\begin{columns}[T]
\begin{column}{0.57\textwidth}
\textbf{In 2020, each municipality has a topic mix:}\\
Some had mostly candidate eligibility challenges; others had more campaign conduct or propaganda violations.
\bigskip

\textbf{By 2024, adversarial litigation waves diffused unevenly across topics.}\\
Waves spread through TRE jurisprudence, party legal strategies, and MPE enforcement patterns --- systemic forces that are state-wide, not municipality-specific.
\bigskip

\textbf{The instrument:}\\
$\bZ$ = municipality's \emph{exposure} to these waves, predicted from its 2020 topic mix $\times$ state-level growth in each topic.\\[4pt]
\footnotesize\textit{How much of the statewide adversarial wave was this municipality structurally positioned to absorb?}
\end{column}
\begin{column}{0.41\textwidth}
\begin{alertblock}{Key Design Logic}
  \begin{enumerate}
    \item \textbf{Shares}: from 2020 --- pre-determined, before 2024 politics materialize
    \item \textbf{Shifts}: state-level waves driven by institutional and strategic diffusion, not by any single municipality's 2024 dynamics
    \item \textbf{Product}: predicted exposure to adversarial litigation waves, exogenous to local competitive outcomes
  \end{enumerate}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Bartik Instrument --- Formal Construction}
\[
\bZ = \sum_{k \,\notin\, \text{admin.}} s_{m,k,2020} \;\cdot\; g_{\text{UF}(m),\,k,\,-m}
\]
\bigskip
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\textwidth}{L{2.4cm}X}
  $s_{m,k,2020}$ & Municipality $m$'s share of topic $k$ among competition-relevant lawsuits in 2020; re-normalized after exclusion of administrative filings so that $\sum_k s_{m,k,2020} = 1$ \\
  $g_{\text{UF},k,-m}$ & Leave-one-out (LOO) log growth of topic $k$ in state UF$(m)$ from 2020 to 2024:
  $\displaystyle g = \log\!\!\left(\frac{T_{\text{UF},k,2024} - C_{m,k,2024}}{T_{\text{UF},k,2020} - C_{m,k,2020}}\right)$ \\
  $k$ & Topic = legal subject matter; competition-relevant subjects only \\
  \text{LOO} & Municipality $m$ excluded from its own state shift to prevent mechanical correlation \\
\end{tabularx}
\end{frame}


\begin{frame}{Topic Selection: Two Instrument Variants}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
We construct two versions of the Bartik instrument, differing only in which case subjects contribute to the shares:
\bigskip
\begin{description}
  \item[\textbf{Broad --- all competition topics:}] Includes all 31 competition-classified subjects from the TSE taxonomy, \emph{including} the mandatory administrative filings (RRC and DRAP). 31 topics, F\,$\approx$\,20.
  \medskip
  \item[\textbf{Narrow --- adversarial only (preferred):}] Excludes mandatory candidacy registration (RRC) and mandatory coalition slate registration (DRAP). Retains 29 genuinely adversarial subjects. F\,$\approx$\,10.
\end{description}
\bigskip
\footnotesize\textit{DRAP exclusion is motivated by the same institutional logic as RRC: it is a mandatory administrative step required by every coalition under Art.~6, Lei~9.504/1997, not a challenge filed against any candidate.}
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Instrument Comparison}
  \footnotesize
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{L{2.2cm}ccc}
    \toprule
    Variant & Topics & Nonzero & F \\
    \midrule
    All topics & 31 & 4,045 & 19.6 \\
    Adversarial & 29 & 4,042 & 10.4 \\
    \bottomrule
  \end{tabular}\\[6pt]
  \normalsize
  Mean Bartik: $-$0.12 / $-$0.14\\
  SD: $\;$0.08 / 0.16
\end{block}
\medskip
\begin{alertblock}{Preferred specification}

\begin{frame}{Shares: 2020 Topic Composition}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Construction:}
\begin{enumerate}
  \item Count unique adversarial cases per (municipality, topic) in 2020
  \item $s_{m,k,2020} = C_{m,k,2020} \,/\, \sum_{k} C_{m,k,2020}$
  \item Shares sum to 1.0 per municipality by construction
  \item Municipalities with zero adversarial 2020 cases: assigned zero instrument value
\end{enumerate}
\bigskip
\textbf{Granularity:} TSE legal subject codes give $\sim$29 adversarial topic cells in the preferred specification
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Why 2020 Shares?}
  Shares are \textbf{pre-determined}: based on the 2020 case distribution, two electoral cycles before the 2024 election dynamics unfold.
  \medskip\\
  Exposure $s_{m,k,2020}$ cannot be chosen or manipulated by actors anticipating 2024 competition --- a core validity argument.
\end{block}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Shifts: State-Level Adversarial Waves}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Construction:}
\begin{enumerate}
  \item Compute state $\times$ topic totals: $T_{\text{UF},k,t}$ for $t \in \{2020, 2024\}$
  \item Remove $m$'s own contribution (LOO): subtract $C_{m,k,t}$
  \item Compute log growth; set undefined if LOO denominator $\leq 0$
\end{enumerate}
\bigskip
\textbf{What the shifts capture:}\\
State-level growth in topic $k$ reflects \emph{waves} of adversarial litigation spreading through the institutional and strategic networks of Brazilian electoral politics --- TRE chamber interpretations, coordinated party legal strategies, and MPE enforcement priorities --- that are common to a state but not driven by any individual municipality.
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Design Choices Defended}
  \begin{description}
    \item[\textbf{State level}] Captures the relevant institutional unit (TRE jurisdiction); closer comparison than national but broader than municipality
    \item[\textbf{LOO}] Prevents $m$ from mechanically driving its own shift
    \item[\textbf{Admin. out}] Adversarial litigation only; mandatory filing noise excluded
    \item[\textbf{Log}] Proportional changes; symmetric treatment of growth and decline
  \end{description}
\end{block}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Exclusion Restriction}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Formal condition:}\\
$E[\bZ \cdot u_m \mid \mathbf{X}_m, \delta_{\text{UF}}] = 0$
\bigskip

\textbf{Core argument --- network diffusion framing:}\\
The instrument does \emph{not} claim that litigation growth is purely random. Rather, $\bZ$ measures a municipality's structural \textbf{exposure} to adversarial litigation waves that originate in and diffuse through the \emph{institutional and strategic networks} of Brazilian electoral politics --- TRE chamber jurisprudence, coordinated party legal strategies, MPE enforcement priorities --- operating at the state level. These systemic forces are orthogonal to the specific competitive environment of any individual municipality, conditional on state FE and pre-determined controls.
\end{column}
\begin{column}{0.43\textwidth}
\begin{alertblock}{Potential Threats}
  \begin{enumerate}
    \item \textbf{Large-municipality dominance}: municipality $m$ drives state shift\\
    $\rightarrow$ mitigated by LOO; most municipalities are small
    \medskip
    \item \textbf{2020 shares reflect 2020 competition}: shares correlated with unobserved persistence\\
    $\rightarrow$ conditional on state FE + 2020 competitive controls
    \medskip
    \item \textbf{Network spillovers}: litigation waves may correlate with state-level political events affecting all municipalities\\
    $\rightarrow$ state FE absorbs common state shocks
  \end{enumerate}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Treatment and Instrument --- Regression Equations}
\textbf{First stage (OLS):}
\[
\underbrace{\Delta\log(1+\ell_{m})}_{\text{endogenous}} \;=\; \alpha + \beta\, \bZ + \mathbf{X}_m'\boldsymbol{\gamma} + \delta_{\text{UF}(m)} + \varepsilon_m
\]
\bigskip
\textbf{Second stage (2SLS):}
\[
\Delta Y_m \;=\; \alpha + \tau\, \widehat{\Delta\log(1+\ell_{m})} + \mathbf{X}_m'\boldsymbol{\gamma} + \delta_{\text{UF}(m)} + u_m
\]
\bigskip
\begin{itemize}
  \item $\ell_m$ = adversarial competition-relevant lawsuits (passive party, pre-election)
  \item $\Delta$ = difference 2024 $-$ 2020 for flow outcomes; level in 2024 for cross-section outcomes
  \item $\mathbf{X}_m$ = pre-determined controls (7 baseline or 14 extended)
  \item $\delta_{\text{UF}(m)}$ = state or microregion fixed effects
  \item SE clustered by principal electoral zone
\end{itemize}
\end{frame}

\section{Conclusion}
% ============================================================

\begin{frame}{Summary of Findings}
\begin{columns}[T]
\begin{column}{0.57\textwidth}
\begin{enumerate}
  \item \textbf{Credible identification:}\\
  Bartik IV valid across two topic-scope variants. Adversarial-only (F\,$\approx$\,10, preferred) and broad (F\,$\approx$\,20). First-stage stable across specifications.
  \bigskip
  \item \textbf{Winner consolidation (directional):}\\
  Both variants point toward more concentrated outcomes: higher winner majority probability, wider margin, lower candidate count. Direction consistent; statistically imprecise at $\alpha=0.05$ with the adversarial-only specification.
  \bigskip
  \item \textbf{Blank vote rate (broad spec only):}\\
  Significant positive effect ($+$1.3 pp, $p=0.018$) under the broader instrument. Not robust in adversarial-only. Suggests voters respond to judicial breadth with expressive abstention.
  \bigskip
  \item \textbf{Voter participation and incumbency:}\\
  Precise nulls for turnout, null rate, incumbent ran/won. New-entrant winner marginally positive in adversarial-only spec ($p=0.08$).
\end{enumerate}
\end{column}
\begin{column}{0.41\textwidth}
\begin{alertblock}{Main Takeaway}
  We provide credibly identified estimates of the electoral effects of judicialization using a Bartik shift-share instrument.
  \medskip\\
  The winner consolidation pattern is directionally consistent and robust to instrument scope. Statistical significance depends on whether mandatory administrative filings are included in the topic mix. This sensitivity is itself informative: administrative litigation breadth --- not just adversarial challenges --- appears to matter for voter behavior.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Contributions and Next Steps}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Contributions:}
\begin{itemize}
  \item First Bartik IV applied to electoral judicialization; establishes a blueprint for identifying causal electoral effects of litigation
  \item Large-scale evidence from 5,560 Brazilian municipalities across two election cycles
  \item Systematic distinction between administrative and adversarial litigation --- a methodological contribution to measurement in the electoral law literature
  \item Two-variant robustness design reveals that administrative filing breadth and adversarial challenges operate through distinct channels
\end{itemize}

    \item Case \emph{outcomes} (ruling for vs.\ against) as a second instrument
    \item Longer panel: 2012--2024 for trend identification

\begin{frame}{Appendix H: Constitutional Threshold Detail}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Art.~29-II, Constituição Federal 1988:}
\begin{quote}
  \small ``eleição do Prefeito e do Vice-Prefeito realizada no dia 1.º de outubro do ano anterior ao do término do mandato dos que devam suceder, aplicadas as regras do art.~77, no caso de Municípios com mais de duzentos mil eleitores.''
\end{quote}
\bigskip
\textbf{Implication:} the 200,000 registered voter threshold determines whether a municipality follows plurality rule (one round) or the two-round runoff rule. This creates a sharp institutional discontinuity that motivates our population subsample as a homogeneity check.
